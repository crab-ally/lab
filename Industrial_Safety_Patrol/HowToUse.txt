# ==================== MuJoCo 시뮬레이터 실행 ====================
1. docker compose up -d
1-1. docker compose exec robot_sim bash

# /cmd_vel 발행 노드
2. docker compose exec robot_sim bash

3. \docker compose --profile slam up -d slam

# ==================== Detection ====================

# 화재감지 (fire_detection_node + fire_fusion_node)

# yolo + ppe + sensor fusion + ttc
7. cd ros2_ws && colcon build --packages-select perception_safety_pkg
7-1. docker compose --profile vision up perception_safety

# 사람의 쓰러짐 감지(Yolo Pose 기반)
