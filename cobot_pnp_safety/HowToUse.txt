# ==========================================
# Franka Panda MoveIt2 3D Vision Pick & Place
# ==========================================

# 1. MuJoCo 시뮬레이션 + MoveIt2 (MoveGroup & Robot State Publisher) 실행
docker compose up mujoco_panda

# 2. RViz2 시각화 실행 (새 터미널)
docker compose --profile rviz up rviz

# 3. 3D 물체 인식 노드 실행 (새 터미널)
docker compose exec mujoco_panda bash
python3 scripts/panda/object_detector_3d.py

# 4. MoveIt 2 기반 Pick & Place 제어 노드 실행 (새 터미널)
docker compose exec mujoco_panda bash
python3 scripts/panda/mujoco_planning_scene.py
python3 scripts/panda/panda_pick_and_place.py

# ==========================================
# 유용한 디버깅 명령어
# ==========================================

# TF 트리 확인:
ros2 run tf2_tools view_frames

# MoveGroup Action 서버 상태 확인:
ros2 action list

# 수동 그리퍼 제어 테스트:
ros2 topic pub --once /panda_gripper_cmd std_msgs/msg/String "{data: 'close'}"
ros2 topic pub --once /panda_gripper_cmd std_msgs/msg/String "{data: 'open'}"
